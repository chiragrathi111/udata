For the Linux :-

Jenkins Install Commands

Step 1 — Install Git

sudo yum update -y
sudo yum install git -y
git --version

Step 2 — Install Java 21

sudo yum install java-21-amazon-corretto -y
java -version

Amazon Linux uses Amazon Corretto (Amazon's own Java) — it is the same as Java 21, fully compatible with Jenkins.

Step 3 — Install Maven

sudo yum install maven -y
mvn --version

Step 4 — Install Jenkins

sudo wget -O /etc/yum.repos.d/jenkins.repo \
    https://pkg.jenkins.io/redhat-stable/jenkins.repo
sudo rpm --import https://pkg.jenkins.io/redhat-stable/jenkins.io-2023.key
sudo yum upgrade -y
sudo yum install jenkins -y

Step 5 — Start Jenkins & Check Status

sudo systemctl enable jenkins
sudo systemctl start jenkins
sudo systemctl status jenkins

Step 6 — Get Admin Password

sudo cat /var/lib/jenkins/secrets/initialAdminPassword
Copy this password and paste it in your browser at http://YOUR_EC2_IP:8080

Search for any package

sudo yum search <software_name>
sudo yum info <software_name>

Docker install :-

sudo apt update -y
sudo apt upgrade -y

sudo apt install -y ca-certificates curl gnupg lsb-release

sudo install -m 0755 -d /etc/apt/keyrings

curl -fsSL https://download.docker.com/linux/ubuntu/gpg | \
sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

sudo chmod a+r /etc/apt/keyrings/docker.gpg

echo \
  "deb [arch=$(dpkg --print-architecture) \
  signed-by=/etc/apt/keyrings/docker.gpg] \
  https://download.docker.com/linux/ubuntu \
  $(. /etc/os-release && echo "$VERSION_CODENAME") stable" | \
  sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

sudo apt update -y

sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

sudo systemctl start docker
sudo systemctl enable docker
sudo systemctl status docker

sudo usermod -aG docker $USER
