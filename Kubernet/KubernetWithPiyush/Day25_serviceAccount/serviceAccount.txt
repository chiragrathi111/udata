# check list Service Account:-
* kubectl get sa
* kubectl get sa -A (All)
* kubectl create sa build-sa  (create a Service Account)
* kubectl describe sa build-sa

# can i check service account have access to getting pod?
* kubectl auth can-i get pods --as build-sa
output:- no

# create a role:-
* kubectl create role build-role \
--verb=list,get,watch \
--resource=pod

# create a role bining and attached role and service account (previously i attached role and user)
* kubectl create rolebinding rb \
--role=build-role \
--user=build-sa
This above command i attached service account instead of user

# can i check service account have access to getting pod?
* kubectl auth can-i get pods --as build-sa
output:- yes

# After that we run get pod :-
* kubectl get pod -A
